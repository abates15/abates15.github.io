You won the 'Halloweekend Quest' and are now a Daily Nebraskan costume expert!"} # The 12th space (index 11) is the finish. ] # --- TIC-TAC-TOE FUNCTIONS --- def get_empty_board(): """Returns a new, empty 3x3 board.""" return [' '] * 9 def print_tictactoe_board(board): """Prints the current Tic-Tac-Toe board to the console.""" print("\n {} | {} | {} ".format(board[0], board[1], board[2])) print(" ---|---|---") print(" {} | {} | {} ".format(board[3], board[4], board[5])) print(" ---|---|---") print(" {} | {} | {} \n".format(board[6], board[7], board[8])) def check_win(board, player): """Checks if the given player has won.""" # Winning combinations (indices of the 1D list) wins = [ (0, 1, 2), (3, 4, 5), (6, 7, 8), # Rows (0, 3, 6), (1, 4, 7), (2, 5, 8), # Columns (0, 4, 8), (2, 4, 6) # Diagonals ] for combo in wins: if all(board[i] == player for i in combo): return True return False def get_available_moves(board): """Returns a list of indices for empty spots.""" return [i for i, spot in enumerate(board) if spot == ' '] def get_player_move(board): """Gets and validates the human player's move.""" while True: try: # User enters a number 1-9 corresponding to the board positions move = input("Enter your move (1-9): ") move_index = int(move) - 1 if 0 <= move_index <= 8 and board[move_index] == ' ': return move_index else: print("Invalid move. Please choose an empty spot (1-9).") except ValueError: print("Invalid input. Please enter a number.") def get_computer_move(board): """Chooses a random available move for the computer.""" available = get_available_moves(board) return random.choice(available) if available else None def play_tictactoe(): """ Plays Tic-Tac-Toe until the human player wins. Returns True when the player wins. """ print("\n--- TIC-TAC-TOE CHALLENGE ---") print("You must win to take your turn! The game repeats until you succeed.") player_token = 'X' computer_token = 'O' while True: board = get_empty_board() current_turn = random.choice(['player', 'computer']) # Random start print(f"\nGame Start! Player is '{player_token}', Computer is '{computer_token}'.") print(f"The first player is: {'You' if current_turn == 'player' else 'The Computer'}") game_over = False while not game_over: print_tictactoe_board(board) if current_turn == 'player': move = get_player_move(board) board[move] = player_token if check_win(board, player_token): print_tictactoe_board(board) print("\n🎉 You won the Tic-Tac-Toe game! You may now roll the dice.") return True # Player wins, exit the TTT loop if not get_available_moves(board): print("\nIt's a draw! You need a win. Starting a new game...") game_over = True current_turn = 'computer' else: # Computer's turn print("Computer is thinking...") time.sleep(1) move = get_computer_move(board) if move is not None: board[move] = computer_token if check_win(board, computer_token): print_tictactoe_board(board) print("\nBoo! The computer won. You need a win to roll. Starting a new game...") game_over = True elif not get_available_moves(board): print("\nIt's a draw! You need a win. Starting a new game...") game_over = True current_turn = 'player' else: # Should only happen on a draw if player didn't already check it game_over = True # --- BOARD GAME FUNCTIONS --- def roll_dice(): """Rolls a random number between 1 and 3.""" return random.randint(1, 3) def print_board(player_pos): """Prints a visual representation of the game board.""" board_str = "\n" # Create the 12 spaces, highlighting the player's position for i in range(len(BOARD_SPACES)): name = BOARD_SPACES[i]['name'] # Format the space if i == 0: display_char = '🏠' elif i == len(BOARD_SPACES) - 1: display_char = '🏆' elif BOARD_SPACES[i]['type'] == 'Fact': display_char = '🧠' elif BOARD_SPACES[i]['type'] == 'Penalty': display_char = '💀' else: display_char = '🟢' # Highlight the player's position if i == player_pos: board_str += f"| {name.upper()} {display_char} *YOU* |" else: board_str += f"| {name} {display_char} |" # Wrap the board for better terminal viewing if (i + 1) % 4 == 0 and i != len(BOARD_SPACES) - 1: board_str += "\n" + "---" * 15 + "\n" elif i != len(BOARD_SPACES) - 1: board_str += " -> " print(board_str) print("-" * 50) print(f"You are currently on Space {player_pos + 1} of {len(BOARD_SPACES)}.") print("-" * 50) def main(): """The main game loop.""" current_position = 0 # 0-indexed position total_spaces = len(BOARD_SPACES) print("=" * 50) print("🎃 HALLOWEEKEND QUEST: A DAILY NEBRASKAN LEARNING GAME 🧠") print("=" * 50) print_board(current_position) # Main game loop while current_position < total_spaces - 1: # 1. The Tic-Tac-Toe Gate (Pre-roll requirement) print("\n[STEP 1: The Tic-Tac-Toe Gate]") play_tictactoe() # This loop only exits when the player wins. # 2. The Dice Roll print("\n[STEP 2: Roll for Movement]") input("Press Enter to roll the 1-3 movement dice...") dice_roll = roll_dice() print(f"The dice rolled a... {dice_roll}! You move {dice_roll} space(s).") # Calculate new position new_position = current_position + dice_roll # Check if the player overshot the finish line if new_position >= total_spaces - 1: current_position = total_spaces - 1 else: current_position = new_position print(f"\nYou moved to Space {current_position + 1}!") print_board(current_position) # 3. Land on a space and resolve action current_space = BOARD_SPACES[current_position] print(f"\n[STEP 3: Landed on: {current_space['name']} ({current_space['type']})]") print(f"ℹ️ {current_space['text']}") if current_space['type'] == 'Penalty': penalty_move = current_space['action'] print(f"Oh no! Penalty: {penalty_move} space(s).") # Apply penalty, ensuring player doesn't go below Start (index 0) current_position = max(0, current_position + penalty_move) print(f"New position after penalty: Space {current_position + 1}") print_board(current_position) elif current_space['type'] == 'Fact': print("You learned a new fact about Daily Nebraskan's costume advice!") input("\n--- Press Enter to continue to the next turn. ---") # Game End print("\n" + "=" * 50) print(f"🎉 {BOARD_SPACES[total_spaces - 1]['text']} 🎉") print("=" * 50) # Standard Python entry point if __name__ == "__main__": try: main() except KeyboardInterrupt: print("\nGame interrupted. Goodbye!") sys.exit(0)
